const LETTERS = ["A", "B", "C", "D"];
const app = document.getElementById("app");
const state = {
  student: null, order: [], current: 0, answers: {},
  startTime: 0, timerId: null, timeSpent: 0, submitted: false,
};

function shuffle(a) {
  const b = a.slice();
  for (let i = b.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [b[i], b[j]] = [b[j], b[i]];
  }
  return b;
}

function fmt(ms) {
  const s = Math.max(0, Math.round(ms / 1000));
  return String(Math.floor(s / 60)).padStart(2, "0") + ":" + String(s % 60).padStart(2, "0");
}

function startQuiz() {
  state.order = shuffle(QUESTIONS.map((_, i) => i)).slice(0, NB_QUESTIONS);
  state.current = 0;
  state.answers = {};
  state.startTime = Date.now();
  state.submitted = false;
  if (DURATION_MINUTES > 0) {
    const deadline = state.startTime + DURATION_MINUTES * 60000;
    state.timerId = setInterval(() => {
      const left = deadline - Date.now();
      const el = document.getElementById("timer");
      if (el) {
        el.textContent = fmt(left);
        if (left < 60000) el.classList.add("warn");
      }
      if (left <= 0) finish(true);
    }, 500);
  }
  renderQuestion();
}

function renderQuestion() {
  const qi = state.order[state.current];
  const q = QUESTIONS[qi];
  const total = state.order.length;
  const progress = Math.round((state.current / total) * 100);
  app.innerHTML = `
    <div class="card">
      <div class="meta">
        <span>Question ${state.current + 1} / ${total}</span>
        ${DURATION_MINUTES > 0 ? '<span class="timer" id="timer">--:--</span>' : "<span>QCM DVI</span>"}
      </div>
      <div class="bar-bg"><div class="bar-fill" style="width:${progress}%"></div></div>
      <h2>${q.q}</h2>
      <div id="choices">${q.choices.map((c, i) => `
        <button type="button" class="choice" data-i="${i}">
          <span class="badge">${LETTERS[i]}</span><span>${c}</span>
        </button>`).join("")}
      </div>
      <button class="btn-primary" id="nextBtn" disabled>
        ${state.current + 1 === total ? "Voir le résultat" : "Question suivante"}
      </button>
    </div>`;
  let sel = null;
  const nextBtn = document.getElementById("nextBtn");
  document.getElementById("choices").addEventListener("click", (e) => {
    const btn = e.target.closest(".choice");
    if (!btn) return;
    sel = Number(btn.dataset.i);
    document.querySelectorAll(".choice").forEach((b) => b.classList.remove("sel"));
    btn.classList.add("sel");
    nextBtn.disabled = false;
  });
  nextBtn.addEventListener("click", () => {
    state.answers[qi] = sel;
    state.current++;
    if (state.current >= total) finish(false);
    else renderQuestion();
  });
}

function finish(timeout) {
  if (state.timerId) { clearInterval(state.timerId); state.timerId = null; }
  state.timeSpent = Date.now() - state.startTime;
  renderResult(timeout);
  if (!state.submitted) submitScore();
}

function renderResult(timeout) {
  const score = state.order.filter((qi) => state.answers[qi] === QUESTIONS[qi].answer).length;
  const total = state.order.length;
  const pct = Math.round((score / total) * 100);
  const emoji = pct >= 80 ? "🎉" : pct >= 50 ? "👍" : "📚";
  const color = pct >= 80 ? "#22c55e" : pct >= 50 ? "#f59e0b" : "#ef4444";
  const review = state.order.map((qi, idx) => {
    const q = QUESTIONS[qi];
    const given = state.answers[qi];
    const ok = given === q.answer;
    return `<div class="review ${ok ? "ok" : "ko"}">
      <p><strong>${idx + 1}. ${q.q}</strong> ${ok ? "✅" : "❌"}</p>
      ${!ok && given != null ? `<p class="bad">Votre réponse : ${LETTERS[given]}. ${q.choices[given]}</p>` : ""}
      <p class="good">Bonne réponse : ${LETTERS[q.answer]}. ${q.choices[q.answer]}</p>
      <p class="exp">${q.explain}</p>
    </div>`;
  }).join("");
  app.innerHTML = `
    <div class="card center">
      ${timeout ? '<div class="alert">⏱ Temps écoulé — le QCM a été clôturé automatiquement.</div>' : ""}
      <div style="font-size:3rem">${emoji}</div>
      <h1 class="scorebig">${score} / ${total}</h1>
      <p class="sub">${state.student.prenom} ${state.student.nom} — ${state.student.service} · ${pct}% · temps : ${fmt(state.timeSpent)}</p>
      <div class="bar-bg" style="margin-top:8px"><div class="bar-fill" style="width:${pct}%;background:${color}"></div></div>
      <div id="submitStatus" class="sub" style="margin-top:6px">${SUBMIT_URL ? "Envoi du score…" : "⚠️ Aucun SUBMIT_URL configuré : le score n'est pas enregistré côté serveur."}</div>
      <div style="height:14px"></div>
      <button class="btn-primary" id="retryBtn" style="max-width:340px">Recommencer (nouveau tirage)</button>
    </div>
    <div style="height:18px"></div>
    ${review}`;
  document.getElementById("retryBtn").addEventListener("click", startQuiz);
}

async function submitScore() {
  state.submitted = true;
  const el = () => document.getElementById("submitStatus");
  const score = state.order.filter((qi) => state.answers[qi] === QUESTIONS[qi].answer).length;
  const payload = {
    nom: state.student.nom, prenom: state.student.prenom, service: state.student.service,
    score: score, total: state.order.length,
    percentage: Math.round((score / state.order.length) * 100),
    time_spent_sec: Math.round(state.timeSpent / 1000),
    date: new Date().toISOString(),
    answers: state.order.map((qi, i) => ({
      question: QUESTIONS[qi].q,
      question_index: qi,
      given: state.answers[qi], correct_index: QUESTIONS[qi].answer,
    })),
  };
  if (!SUBMIT_URL) return;
  try {
    const res = await fetch(SUBMIT_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    if (!res.ok) throw new Error("HTTP " + res.status);
    if (el()) el().innerHTML = '<span class="okmsg" style="display:block">✅ Score enregistré avec succès.</span>';
  } catch (err) {
    if (el()) el().innerHTML = '<div class="alert" style="display:block">⚠️ Échec de l\'envoi du score (' + err.message + '). Contactez votre enseignant.</div>';
  }
}

renderIntro();
